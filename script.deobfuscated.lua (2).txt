local function helper19()
  local getMouseLocation = userInputService:GetMouseLocation()

  local viewportPointToRay = workspace.CurrentCamera:ViewportPointToRay(
    getMouseLocation.X, getMouseLocation.Y
  )

  local raycastParams = RaycastParams.new()
  raycastParams.FilterType = Enum.RaycastFilterType.Exclude

  local character11 = localPlayer.Character

  if character11 then
    raycastParams.FilterDescendantsInstances = { character11 }
  end

  local raycast = workspace:Raycast(
    viewportPointToRay.Origin, viewportPointToRay.Direction * 1000, raycastParams
  )

  if raycast and raycast.Instance then
    local instance2 = raycast.Instance

    while instance2 and not instance2:FindFirstChildOfClass("Humanoid") do
      instance2 = instance2.Parent

      if instance2 == workspace then
        instance2 = nil
        break
      end
    end

    if instance2 then
      for index31, value36 in ipairs(players:GetPlayers()) do
        if value36.Character == instance2 and value36 ~= localPlayer then
          return value36
        end
      end
    end
  end

  local val93, val94 = nil, 15

  for index32, value37 in ipairs(players:GetPlayers()) do
    if value37 ~= localPlayer and value37.Character then
      local humanoidRootPart2 = value37.Character:FindFirstChild("HumanoidRootPart")

      if humanoidRootPart2 then
        local dot = (humanoidRootPart2.Position - viewportPointToRay.Origin):Dot(viewportPointToRay.Direction)

        if dot > 0 then
          local magnitude2 = (humanoidRootPart2.Position
            - (viewportPointToRay.Origin + viewportPointToRay.Direction * dot)).Magnitude

          if magnitude2 < val94 then
            val94 = magnitude2
            val93 = value37
          end
        end
      end
    end
  end

  return val93
end
